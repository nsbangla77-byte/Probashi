const express = require("express"); const session = require("express-session"); const rateLimit = require("express-rate-limit"); const helmet = require("helmet"); const bcrypt = require("bcryptjs"); const Database = require("better-sqlite3"); const path = require("path"); const app = express(); const db = new Database("data.db"); const PORT = process.env.PORT || 3000; db.exec(` CREATE TABLE IF NOT EXISTS admins ( id INTEGER PRIMARY KEY AUTOINCREMENT, username TEXT UNIQUE NOT NULL, password_hash TEXT NOT NULL ); CREATE TABLE IF NOT EXISTS links ( id INTEGER PRIMARY KEY AUTOINCREMENT, name TEXT NOT NULL, url TEXT NOT NULL, icon TEXT NOT NULL DEFAULT 'globe', created_at TEXT DEFAULT CURRENT_TIMESTAMP ); `); const adminUser = process.env.ADMIN_USERNAME || "admin"; const adminPass = process.env.ADMIN_PASSWORD; if (!adminPass) { console.error("Set ADMIN_PASSWORD as a Replit Secret before running."); process.exit(1); } const existing = db.prepare("SELECT id FROM admins WHERE username=?").get(adminUser); if (!existing) { const hash = bcrypt.hashSync(adminPass, 12); db.prepare("INSERT INTO admins(username,password_hash) VALUES(?,?)").run(adminUser, hash); } if (db.prepare("SELECT COUNT(*) c FROM links").get().c === 0) { db.prepare("INSERT INTO links(name,url,icon) VALUES(?,?,?)") .run("প্রবাসী ও যুব কল্যাণ সমবায় সমিতি", "https://example.com", "globe"); } app.use(helmet({ contentSecurityPolicy: { directives: { defaultSrc: ["'self'"], styleSrc: ["'self'", "https://cdn.tailwindcss.com"], scriptSrc: ["'self'", "https://cdn.tailwindcss.com"], imgSrc: ["'self'", "data:"], connectSrc: ["'self'"], fontSrc: ["'self'"], objectSrc: ["'none'"], baseUri: ["'self'"], frameAncestors: ["'none'"] } } })); app.use(express.json({ limit: "20kb" })); app.use(express.urlencoded({ extended: false, limit: "20kb" })); app.use(session({ secret: process.env.SESSION_SECRET || "CHANGE_THIS_SECRET_IN_REPLIT", resave: false, saveUninitialized: false, cookie: { httpOnly: true, sameSite: "strict", secure: process.env.NODE_ENV === "production", maxAge: 30 * 60 * 1000 } })); app.use(express.static(path.join(__dirname, "public"))); const loginLimiter = rateLimit({ windowMs: 15 * 60 * 1000, max: 8, standardHeaders: true, legacyHeaders: false }); function adminOnly(req, res, next) { if (req.session.adminId) return next(); res.status(401).json({ error: "অনুমতি নেই" }); } function validUrl(value) { try { const u = new URL(value); return ["http:", "https:"].includes(u.protocol); } catch { return false; } } app.post("/api/login", loginLimiter, (req, res) => { const { username, password } = req.body; const admin = db.prepare("SELECT * FROM admins WHERE username=?").get(username || ""); if (!admin || !bcrypt.compareSync(password || "", admin.password_hash)) { return res.status(401).json({ error: "ভুল লগইন তথ্য" }); } req.session.adminId = admin.id; res.json({ ok: true }); }); app.post("/api/logout", (req, res) => { req.session.destroy(() => res.json({ ok: true })); }); app.get("/api/links", (req, res) => { res.json(db.prepare("SELECT id,name,url,icon FROM links ORDER BY id DESC").all()); }); app.post("/api/links", adminOnly, (req, res) => { const { name, url, icon } = req.body; if (!name || name.length > 120 || !validUrl(url)) { return res.status(400).json({ error: "নাম বা লিংক সঠিক নয়" }); } const result = db.prepare("INSERT INTO links(name,url,icon) VALUES(?,?,?)") .run(name.trim(), url.trim(), (icon || "globe").slice(0, 30)); res.json({ id: result.lastInsertRowid }); }); app.delete("/api/links/:id", adminOnly, (req, res) => { db.prepare("DELETE FROM links WHERE id=?").run(Number(req.params.id)); res.json({ ok: true }); }); app.get("/api/me", (req, res) => res.json({ admin: !!req.session.adminId })); app.listen(PORT, () => console.log(`Running on port ${PORT}`));
